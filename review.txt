import React, { useState } from 'react';

export default function OrderModal({ order, onClose, onUpdate, onDelete }) {
  // State for all editable fields
  const [status, setStatus] = useState(order.status);
  const [type, setType] = useState(order.type);
  const [items, setItems] = useState(order.items);
  const [price, setPrice] = useState(order.price); // <--- NEW: Price state

  const handleSave = () => {
    // Send all fields back to App.jsx
    onUpdate(order.id, { ...order, status, type, items, price: parseFloat(price) });
  };

  return (
    <div className="modal-overlay">
      <div className="modal-content">
        
        <div className="modal-header">
          <h2>Edit Order #{order.id}</h2>
          <button className="close-btn" onClick={onClose}>&times;</button>
        </div>

        <div className="form-group">
          <label>Order Type (Table, Takeout, Delivery)</label>
          <input type="text" value={type} onChange={(e) => setType(e.target.value)} />
        </div>

        <div className="form-group">
          <label>Items</label>
          <input type="text" value={items} onChange={(e) => setItems(e.target.value)} />
        </div>

        {/* NEW: Price Input Field */}
        <div className="form-group">
          <label>Price ($)</label>
          <input 
            type="number" 
            step="0.01" 
            value={price} 
            onChange={(e) => setPrice(e.target.value)} 
          />
        </div>

        <div className="form-group">
          <label>Status</label>
          <select value={status} onChange={(e) => setStatus(e.target.value)}>
            <option value="Pending">Pending</option>

            <option value="Completed">Completed</option>
          </select>
        </div>

        <div className="modal-actions">
          <button className="btn btn-danger" onClick={() => onDelete(order.id)}>
            Delete Order

          <button className="btn btn-primary" onClick={handleSave}>
            Save Changes
          </button>
        </div>

      </div>
    </div>
  );
}

import './App.css';

function App() {
  const [searchQuery, setSearchQuery] = useState('');
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [currentPage, setCurrentPage] = useState('kitchen');
  
  const [ordersData, setOrdersData] = useState([]);
  const [historyData, setHistoryData] = useState([]); 
  const [selectedOrder, setSelectedOrder] = useState(null); 

  const fetchOrders = () => {
    fetch('http://localhost:8080/api/orders')
      .then(res => res.json())
      .then(data => setOrdersData(data))
      .catch(err => console.error("Error fetching:", err));
  };

  const fetchHistory = () => {
    fetch('http://localhost:8080/api/orders/history')
      .then(res => res.json())
      .then(data => setHistoryData(data))
      .catch(err => console.error("Error fetching history:", err));
  };

  useEffect(() => {
    if (isLoggedIn) {
      fetchOrders(); 
      if (currentPage === 'history') {
        fetchHistory(); 
      }
    }
  }, [isLoggedIn, currentPage]); 

  const handleUpdateOrder = (id, updatedOrder) => {
    fetch(`http://localhost:8080/api/orders/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(updatedOrder)
    })
    .then(res => res.json())
    .then(() => {
      setOrdersData(ordersData.map(o => o.id === id ? updatedOrder : o));
      
      if (currentPage === 'history') {
        fetchHistory();
      }
      if (selectedOrder) setSelectedOrder(null); 
    });
  };

  const handleDeleteOrder = (id) => {
    if(window.confirm("Are you sure you want to delete this order?")) {
      fetch(`http://localhost:8080/api/orders/${id}`, { method: 'DELETE' })
      .then(() => {
        setOrdersData(ordersData.filter(o => o.id !== id));
        setHistoryData(historyData.filter(o => o.id !== id)); 
        setSelectedOrder(null); 
      });
    }
  };

  const handleNewOrder = () => {
    const newOrder = { type: 'Takeout', items: '1x Mystery Pizza', price: 19.99, status: 'Pending' };
    fetch('http://localhost:8080/api/orders', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(newOrder)
    })
    .then(res => res.json())
    .then(savedOrder => setOrdersData([...ordersData, savedOrder]));
  };

  const totalRevenue = ordersData.reduce((sum, order) => sum + order.price, 0);
  
  // --- SEARCH FILTER LOGIC START ---
  const filteredOrders = ordersData.filter(order => 
    order.id.toString().includes(searchQuery.toLowerCase()) || 
    order.items.toLowerCase().includes(searchQuery.toLowerCase()) || 
    order.type.toLowerCase().includes(searchQuery.toLowerCase())
  );

  const pendingOrders = filteredOrders.filter(o => o.status === 'Pending');
  const cookingOrders = filteredOrders.filter(o => o.status === 'Cooking');
  const completedOrders = filteredOrders.filter(o => o.status === 'Completed');
  // --- SEARCH FILTER LOGIC END ---

  if (!isLoggedIn) {
    return (
      <div id="app" className="theme-black">
        <Login onLogin={() => setIsLoggedIn(true)} />
      </div>
    );
  }

  return (
    <div id="app" className="theme-black dashboard-layout">
      
      <aside className="sidebar">
        <div className="sidebar-logo">ðŸ• RestaurantOS</div>
        <nav className="sidebar-nav">
          <a 
            href="#" 
            className={currentPage === 'kitchen' ? 'active' : ''} 
            onClick={(e) => { e.preventDefault(); setCurrentPage('kitchen'); }}
          >
            <i className="fa fa-fire"></i> Kitchen Display
          </a>
          
          <a 
            href="#" 
            className={currentPage === 'history' ? 'active' : ''} 
            onClick={(e) => { e.preventDefault(); setCurrentPage('history'); }}
          >
            <i className="fa fa-history"></i> Order History
          </a>

package com.restaurant.restaurant_api;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;
import java.util.List;

@RestController
@RequestMapping("/api/orders")
@CrossOrigin(origins = "http://localhost:5173")
public class OrderController {

    @Autowired
    private OrderRepository orderRepository;

    // GET: Send all orders to React
    @GetMapping
    public List<Order> getAllOrders() {
        return orderRepository.findAll();
    }

    // GET: Send only completed orders to React
    @GetMapping("/history")
    public List<Order> getOrderHistory() {
        return orderRepository.findByStatus("Completed");
    }

    // POST: Receive a new order from React and save it
    @PostMapping
    public Order createOrder(@RequestBody Order order) {
        return orderRepository.save(order);
    }

    // PUT: Update an order
    @PutMapping("/{id}")
    public Order updateOrder(@PathVariable Long id, @RequestBody Order orderDetails) {
        Order order = orderRepository.findById(id).orElseThrow();
        
        order.setStatus(orderDetails.getStatus());
        order.setType(orderDetails.getType());
        order.setItems(orderDetails.getItems());
        order.setPrice(orderDetails.getPrice());
        
        return orderRepository.save(order);
    }

    // DELETE: Remove an order from the database
    @DeleteMapping("/{id}")
    public void deleteOrder(@PathVariable Long id) {
        orderRepository.deleteById(id);
    }
} // <--- This is the closing brace for the class that was missing!
